博客文章

终于到了“博客”这部分。这一章让用户能发文章、看文章列表， 并解决两个每个内容型网站都要面对的问题：分页和富文本。 富文本那一节是整本教程安全性最微妙的地方 —— 我们要让用户写 HTML，又不能让他们注入脚本。

package com.example.javalinky.model;

import com.example.javalinky.util.Markdown;
import jakarta.persistence.*;
import java.time.Instant;

@Entity
@Table(name = "posts")
public class Post {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /** 用户写的原文（Markdown） */
    @Column(nullable = false, columnDefinition = "text")
    private String body;

    /** 渲染并清洗之后的 HTML，见本章"富文本"一节 */
    @Column(name = "body_html", columnDefinition = "text")
    private String bodyHtml;

    @Column(name = "timestamp", nullable = false)
    private Instant timestamp = Instant.now();

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "author_id", nullable = false)
    private User author;

    @OneToMany(mappedBy = "post", fetch = FetchType.LAZY,
               cascade = CascadeType.ALL, orphanRemoval = true)
    private java.util.List<Comment> comments = new java.util.ArrayList<>();

    protected Post() {}

    public Post(String body, User author) {
        this.body = body;
        this.author = author;
    }

    /**
     * JPA 生命周期回调：在 INSERT 和 UPDATE 之前自动渲染 Markdown。
     * 相当于原书用 SQLAlchemy 的 db.event.listen(Post.body, 'set', ...) 做的事。
     */
    @PrePersist
    @PreUpdate
    private void renderBody() {
        this.bodyHtml = Markdown.toSafeHtml(body);
    }

    public Long getId() { return id; }
    public String getBody() { return body; }
    public void setBody(String body) { this.body = body; }
    public String getBodyHtml() { return bodyHtml; }
    public Instant getTimestamp() { return timestamp; }
    public User getAuthor() { return author; }
    public java.util.List<Comment> getComments() { return comments; }
}
☀ @PrePersist / @PreUpdate 就是 SQLAlchemy 的事件监听

原书用 db.event.listen(Post.body, 'set', Post.on_changed_body)—— 每次给 body 赋值就自动更新 body_html。

JPA 的生命周期回调时机稍有不同：它在写数据库之前触发，而不是赋值时。 实际效果一样，而且更好 —— 哪怕中间改了十次 body，也只渲染一次。

可用的回调还有：@PostPersist、@PostUpdate、 @PreRemove、@PostRemove、@PostLoad。

package com.example.javalinky.db;

import com.example.javalinky.model.Post;
import com.example.javalinky.model.User;
import jakarta.persistence.EntityManager;
import java.util.Optional;

public class PostRepository {

    private final EntityManager em;

    public PostRepository(EntityManager em) {
        this.em = em;
    }

    public Post save(Post post) {
        em.persist(post);
        return post;
    }

    public Optional<Post> findById(long id) {
        return Optional.ofNullable(em.find(Post.class, id));
    }

    public long countByAuthor(User author) {
        return em.createQuery(
                "select count(p) from Post p where p.author = :a", Long.class)
            .setParameter("a", author)
            .getSingleResult();
    }

    /** 全部文章，按时间倒序分页。join fetch 作者，避免列表页 N+1 */
    public Page<Post> findAll(int page, int size) {
        long total = em.createQuery("select count(p) from Post p", Long.class)
            .getSingleResult();

        var items = em.createQuery("""
                select p from Post p
                  join fetch p.author a
                 order by p.timestamp desc
                """, Post.class)
            .setFirstResult((page - 1) * size)
            .setMaxResults(size)
            .getResultList();

        return Page.of(items, page, size, total);
    }

    public Page<Post> findByAuthor(User author, int page, int size) {
        long total = countByAuthor(author);
        var items = em.createQuery("""
                select p from Post p
                  join fetch p.author a
                 where p.author = :author
                 order by p.timestamp desc
                """, Post.class)
            .setParameter("author", author)
            .setFirstResult((page - 1) * size)
            .setMaxResults(size)
            .getResultList();
        return Page.of(items, page, size, total);
    }
}

分页

原书用 Flask-SQLAlchemy 的 paginate() 方法，它返回一个 Pagination 对象，带着 items、page、 has_prev、has_next、iter_pages() 等等。 JPA 没有这个东西，我们自己做一个 —— 大概 50 行： 

package com.example.javalinky.db;

import java.util.ArrayList;
import java.util.List;

/** 一页数据 + 分页元信息。相当于 Flask-SQLAlchemy 的 Pagination。 */
public record Page<T>(List<T> items, int page, int size, long total) {

    public static <T> Page<T> of(List<T> items, int page, int size, long total) {
        return new Page<>(items, page, size, total);
    }

    public static <T> Page<T> empty(int size) {
        return new Page<>(List.of(), 1, size, 0);
    }

    public int totalPages() {
        return total == 0 ? 1 : (int) Math.ceil((double) total / size);
    }

    public boolean hasPrevious() { return page > 1; }
    public boolean hasNext()     { return page < totalPages(); }
    public int previousPage()    { return Math.max(1, page - 1); }
    public int nextPage()        { return Math.min(totalPages(), page + 1); }
    public boolean isEmpty()     { return items.isEmpty(); }

    /**
     * 要显示哪些页码。0 表示省略号。
     * 相当于 Flask-SQLAlchemy 的 iter_pages()。
     * 例如当前第 7 页、共 20 页 → [1, 0, 5, 6, 7, 8, 9, 0, 20]
     */
    public List<Integer> pageNumbers() {
        int last = totalPages();
        int window = 2;                     // 当前页左右各显示几个
        List<Integer> result = new ArrayList<>();
        int previous = 0;

        for (int i = 1; i <= last; i++) {
            boolean show = i <= 1                       // 第一页总是显示
                        || i > last - 1                  // 最后一页总是显示
                        || Math.abs(i - page) <= window; // 当前页附近
            if (!show) continue;
            if (previous != 0 && i - previous > 1) result.add(0);  // 中间有断档 → 省略号
            result.add(i);
            previous = i;
        }
        return result;
    }

    /** 把页码参数规整到合法范围，防止 ?page=-5 或 ?page=99999 */
    public static int clampPage(Integer requested) {
        if (requested == null || requested < 1) return 1;
        return Math.min(requested, 100_000);
    }
}

🔎 深入：OFFSET 分页为什么会越翻越慢

limit 20 offset 100000 并不是"直接跳到第 10 万条"—— 数据库必须先扫出前 10 万零 20 条，再把前 10 万条丢掉。 所以第 1 页毫秒级，第 5000 页可能要几秒。

另外还有一个正确性问题：你在看第 2 页的时候，有人在第 1 页插入了一条新文章， 于是原本在第 1 页末尾的那条被"挤"到了第 2 页开头 —— 你会看到重复的条目。

游标分页（keyset pagination）能解决这两个问题： 不用页码，而是记住"上一页最后一条的时间戳和 id"： 

-- 第一页
select * from posts order by timestamp desc, id desc limit 20;

-- 下一页：带上上一页最后一条的值
select * from posts
 where (timestamp, id) < (:lastTimestamp, :lastId)
 order by timestamp desc, id desc
 limit 20;

配上 ix_posts_timestamp 索引，每一页的耗时都一样。 代价是没法"跳到第 100 页"（只能一页页往后翻）—— 这也是为什么 社交网站都是"无限滚动"而不是页码。

本教程用 OFFSET 分页（跟原书一致，而且有页码控件好演示）， 但数据量上来后请改成游标分页。

文章列表也做成页面模型，顺便把每条的权限判断算好：

package com.example.javalinky.web.page;

import com.example.javalinky.db.Page;
import com.example.javalinky.model.Permission;
import com.example.javalinky.model.Post;
import com.example.javalinky.model.User;
import java.time.Instant;
import java.util.List;

public record PostListPage(Page<Item> page) {

    public record Item(
        long id,
        String bodyHtml,
        Instant timestamp,
        String authorUsername,
        String authorAvatarHash,
        long commentCount,
        boolean canEdit        // 当前访客能不能编辑这一篇
    ) {}

    public static PostListPage of(Page<Post> source, User viewer) {
        List<Item> items = source.items().stream()
            .map(post -> new Item(
                post.getId(),
                post.getBodyHtml(),
                post.getTimestamp(),
                post.getAuthor().getUsername(),
                post.getAuthor().avatarHash(),
                post.getComments().size(),      // 注意：这会触发懒加载，见下面的说明
                canEdit(viewer, post)))
            .toList();

        return new PostListPage(
            Page.of(items, source.page(), source.size(), source.total()));
    }

    private static boolean canEdit(User viewer, Post post) {
        if (viewer == null) return false;
        return viewer.getId().equals(post.getAuthor().getId())
            || viewer.can(Permission.ADMIN);
    }
}
⚠ post.getComments().size() 是一个 N+1 陷阱

列 20 篇文章，这行代码会发 20 条 select … from comments where post_id = ?。 更糟的是它把每条评论的全部字段都查出来了，只为了数个数。

正确做法是用一条聚合查询把评论数一次查完：

/** 批量查"每篇文章有多少条评论" */
public Map<Long, Long> countCommentsByPost(List<Long> postIds) {
    if (postIds.isEmpty()) return Map.of();
    return em.createQuery("""
            select c.post.id, count(c)
              from Comment c
             where c.post.id in :ids
             group by c.post.id
            """, Object[].class)
        .setParameter("ids", postIds)
        .getResultStream()
        .collect(java.util.stream.Collectors.toMap(
            row -> (Long) row[0],
            row -> (Long) row[1]));
}

造假数据

分页在只有 3 篇文章时根本测不出来。原书用 Python 的 Faker 库造了 100 个用户和 100 篇文章； Java 的对应物是 Datafaker（Java Faker 的继任者）。

  <groupId>net.datafaker</groupId>
  <artifactId>datafaker</artifactId>

</dependency>

import net.datafaker.Faker;
import java.time.temporal.ChronoUnit;
import java.util.Locale;
import java.util.Random;

/** 造假数据。只在开发环境用。 */
public static void fake(int userCount, int postCount) {
    var faker = new Faker(Locale.SIMPLIFIED_CHINESE);
    var random = new Random();
    var defaultRole = new RoleRepository(em).defaultRole();

    // --- 用户 ---
    int created = 0;
    while (created < userCount) {
        String email = faker.internet().emailAddress();
        String username = faker.internet().username();

        // 假数据会撞车，撞了就跳过
        if (exists(email, username)) continue;

        var user = new User(email, username);
        user.setPassword("password");          // 开发用，统一密码方便登录测试
        user.setConfirmed(true);
        user.setName(faker.name().fullName());
        user.setLocation(faker.address().city());
        user.setAboutMe(faker.lorem().sentence(12));
        user.setRole(defaultRole);
        em.persist(user);
        created++;

        // 每 50 条 flush 一次，别让一级缓存无限膨胀
        if (created % 50 == 0) {
            em.flush();
            em.clear();
            defaultRole = new RoleRepository(em).defaultRole();  // clear 之后要重新取
        }
    }
    em.flush();

    // --- 文章 ---
    var userIds = em.createQuery("select u.id from User u", Long.class).getResultList();
    for (int i = 0; i < postCount; i++) {
        var author = em.getReference(User.class, userIds.get(random.nextInt(userIds.size())));
        var post = new Post(fakeMarkdown(faker), author);
        // 随机散布在过去一年里，这样时间排序才有意义
        post.setTimestamp(Instant.now().minus(random.nextInt(365 * 24), ChronoUnit.HOURS));
        em.persist(post);
        if (i % 50 == 0) em.flush();
    }

    System.out.printf("已生成 %d 个用户、%d 篇文章%n", created, postCount);
}

/** 带点 Markdown 语法的假正文，方便验证渲染 */
private static String fakeMarkdown(Faker faker) {
    return """
        ## %s

        %s

        - %s
        - %s

        用 `代码` 和 **粗体** 以及 [链接](https://example.com) 测试一下渲染。
        """.formatted(
            faker.lorem().sentence(5),
            faker.lorem().paragraph(4),
            faker.lorem().sentence(6),
            faker.lorem().sentence(6));
}

💡 em.flush() + em.clear() 为什么必要

EntityManager 的一级缓存会一直持有它管理过的所有实体。 批量插入 10 万条数据时，这些对象全在内存里，而且每次 flush 的脏检查 都要遍历全部对象 —— 结果是越来越慢，最后 OutOfMemoryError。

flush() 把挂起的 SQL 发出去，clear() 清空缓存。 代价是清空之后之前拿到的实体都变成游离状态， 所以上面的代码要重新取 defaultRole。

富文本：Markdown + 标签白名单

现在文章只能是纯文本。要让用户写标题、列表、链接、代码块，有三条路：

方案做法风险
直接存 HTML给用户一个富文本编辑器，存它产生的 HTML极高：必须彻底清洗，稍有疏漏就是 XSS
自定义标记语言如论坛的 [b]粗体[/b]低，但用户要学一套没人用的语法
Markdown用户写 Markdown，服务端渲染成 HTML中：渲染结果仍需清洗（Markdown 允许内嵌 HTML）

原书和本教程都选 Markdown。处理链路是这样的：

package com.example.javalinky.util;

import org.commonmark.ext.gfm.tables.TablesExtension;
import org.commonmark.parser.Parser;
import org.commonmark.renderer.html.HtmlRenderer;
import org.owasp.html.HtmlPolicyBuilder;
import org.owasp.html.PolicyFactory;

import java.util.List;

public final class Markdown {

    private static final Parser PARSER = Parser.builder()
        .extensions(List.of(TablesExtension.create()))
        .build();

    private static final HtmlRenderer RENDERER = HtmlRenderer.builder()
        .extensions(List.of(TablesExtension.create()))
        .build();

    /**
     * 允许的标签白名单。原则是：只放进展示文章确实需要的标签，
     * 其他一律删掉。不要写黑名单 —— 你永远列不全攻击者的花样。
     */
    private static final PolicyFactory POLICY = new HtmlPolicyBuilder()
        .allowElements(
            "p", "br", "hr",
            "h1", "h2", "h3", "h4", "h5", "h6",
            "strong", "em", "b", "i", "s", "del",
            "ul", "ol", "li",
            "blockquote", "code", "pre",
            "a", "table", "thead", "tbody", "tr", "th", "td")
        // 只允许这几种链接协议 —— 挡掉 javascript: 和 data:
        .allowUrlProtocols("http", "https", "mailto")
        .allowAttributes("href").onElements("a")
        // 外链加上 rel="nofollow noopener"，防止被利用做 SEO 和窗口劫持
        .requireRelNofollowOnLinks()
        // 代码块的语言标记（<code class="language-java">）
        .allowAttributes("class").matching(
            java.util.regex.Pattern.compile("^language-[a-z0-9+#-]+$")).onElements("code")
        .toFactory();

    /** 文章用：渲染 Markdown 然后清洗 */
    public static String toSafeHtml(String markdown) {
        if (markdown == null || markdown.isBlank()) return "";
        String dirty = RENDERER.render(PARSER.parse(markdown));
        return POLICY.sanitize(dirty);
    }

    /** 评论用：标签更少（第 13 章） */
    private static final PolicyFactory COMMENT_POLICY = new HtmlPolicyBuilder()
        .allowElements("p", "br", "strong", "em", "b", "i", "code", "a")
        .allowUrlProtocols("http", "https", "mailto")
        .allowAttributes("href").onElements("a")
        .requireRelNofollowOnLinks()
        .toFactory();

    public static String toSafeCommentHtml(String markdown) {
        if (markdown == null || markdown.isBlank()) return "";
        return COMMENT_POLICY.sanitize(RENDERER.render(PARSER.parse(markdown)));
    }

    private Markdown() {}
}
☀ 白名单的三条铁律
  1. 用白名单，不用黑名单。 "禁止 <script>" 这种思路必败 —— 还有 <img onerror=>、<svg onload=>、 <iframe srcdoc=>、<style> 里的 expression()……列不完的。
  2. 属性也要白名单。 允许 <a> 但不限制属性，用户就能写 <a onmouseover="...">。
  3. URL 协议也要白名单。 <a href="javascript:alert(1)"> 是一个经典绕过。 只允许 http / https / mailto。

⚠ 客户端预览和服务端渲染可能不一致

EasyMDE 用 marked.js 做预览，服务端用 commonmark-java。 两者对边缘语法的处理可能有细微差别，更重要的是 —— 预览里没有白名单清洗，所以用户在预览里看到的 <iframe> 提交后会消失。

彻底一致的做法是：预览也走服务端（提交一个 AJAX 请求到 /preview， 返回清洗后的 HTML）。多一次往返，但所见即所得。

🔎 深入：为什么要存 body_html，而不是每次渲染

另一种设计是只存 Markdown 原文，每次展示时渲染。两种做法的对比：

存渲染结果（本教程）每次渲染
读性能好：直接输出字符串差：列 20 篇要渲染 20 次
存储多占一倍空间省
改白名单后麻烦：要写脚本重新渲染全部历史数据自动生效
修了渲染器的 bug同上，需要回填自动生效
安全边界在写入时（一次）在读取时（每次）

原书选了"存渲染结果"，我们跟随。 但要记住：一旦你改了白名单或升级了渲染器，必须准备一个回填任务： 

// Cli 里加一个命令：rerender
em.createQuery("select p from Post p", Post.class)
  .getResultStream()
  .forEach(p -> p.setBody(p.getBody()));   // 触发 @PreUpdate 重新渲染

折中方案：存渲染结果 + 带上渲染器版本号， 读取时发现版本旧了就重新渲染并更新。这样回填是渐进发生的，不需要停机跑脚本。

/** 单篇文章页。URL: /post/{id} */
private void post(Context ctx) {
    long id = ctx.pathParamAsClass("id", Long.class).get();
    Post post = services.postRepo(ctx).findById(id)
        .orElseThrow(() -> new NotFoundResponse("文章不存在"));

    // 第 13 章会在这里加上评论列表
    View.render(ctx, "post.jte", "文章",
                "post", PostListPage.of(
                    Page.of(List.of(post), 1, 1, 1), CurrentUser.get(ctx)));
}

private void editPostForm(Context ctx) {
    Post post = loadEditablePost(ctx);
    View.render(ctx, "edit_post.jte", "编辑文章",
                "form", new FormView(Map.of("body", post.getBody()), Map.of()),
                "postId", post.getId());
}

private void editPost(Context ctx) {
    Post post = loadEditablePost(ctx);
    var form = new PostForm(ctx.formParam("body"));

    var errors = Forms.validate(form);
    if (!errors.isEmpty()) {
        View.render(ctx, "edit_post.jte", "编辑文章",
                    "form", new FormView(Map.of("body", nullToEmpty(form.body())), errors),
                    "postId", post.getId());
        return;
    }

    post.setBody(form.body());     // @PreUpdate 会自动重新渲染 bodyHtml
    Flash.success(ctx, "文章已更新。");
    ctx.redirect(Routes.post(post.getId()));
}

/**
 * 取出文章并检查"当前用户能不能编辑它"。
 * 注意：路由上的 Permission.WRITE 只保证"这个人有发文章的权限"，
 * 不保证"这篇文章是他的" —— 后者必须在这里查。
 */
private Post loadEditablePost(Context ctx) {
    User user = CurrentUser.require(ctx);
    long id = ctx.pathParamAsClass("id", Long.class).get();
    Post post = services.postRepo(ctx).findById(id)
        .orElseThrow(() -> new NotFoundResponse("文章不存在"));

    boolean isAuthor = user.getId().equals(post.getAuthor().getId());
    if (!isAuthor && !user.can(Permission.ADMIN)) {
        throw new ForbiddenResponse("你不能编辑别人的文章");
    }
    return post;
}
☀ 两种授权，都不能少
检查什么在哪做
基于角色"你有编辑文章这个功能的权限吗"路由上的 RouteRole（第 9 章）
基于归属"这个具体的对象是你的吗"处理器里，取出对象之后

漏掉第二种是 Web 应用最常见的严重漏洞之一，OWASP 把它叫做 IDOR（不安全的直接对象引用）： 攻击者把 URL 里的 /edit/5 改成 /edit/6 就改了别人的数据。

任何时候 URL 或表单里出现了对象 id，都要问一句："这个 id 指向的东西，当前用户有权动吗？"
